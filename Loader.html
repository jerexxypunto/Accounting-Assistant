class GSC_Loader { constructor(target) { this.html = `
Loading...
Loading...
Loading...

Cargando...

`; this.parentElement = document.querySelector(target); } addLoader() { if (!this.parentElement) { return null; } const mount = document.createElement("div"); mount.className = "gsc-flow-loader-mount"; mount.innerHTML = this.html; this.parentElement.appendChild(mount); return this.parentElement.querySelector(".gsc-flow-loader-mount .loader-container"); } toggle() { const loader = this.parentElement?.querySelector(".loader-container"); if (loader) { loader.classList.toggle("d-none"); } } }
